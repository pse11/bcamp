--처음 설정
SQL> sqlplus
SQL> /as sysdba
SQL> ALTER SESSION SET "_ORACLE_SCRIPT"=TRUE;

세션이 변경되었습니다.

SQL> COMMIT;

커밋이 완료되었습니다.

-- 계정 생성 사용자명 : MULTI, PW: MULTI로 계정 만들겠다
SQL> CREATE USER MULTI IDENTIFIED BY MULTI;

사용자가 생성되었습니다.

--권한 부여
SQL> GRANT CONNECT, RESOURCE TO MULTI;

권한이 부여되었습니다.

-- 데이터 입력 가능하게 설정
SQL> GRANT UNLIMITED TABLESPACE TO MULTI;

권한이 부여되었습니다.

--시스템 계정-> MULTI 계정 접속
SQL> CONN MULTI/MULTI;

연결되었습니다.

--접속 방법
SQL> sqlplus
SQL> MULTI
SQL> MULTI

-- 테이블 생성
    
SQL> CREATE TABLE TEST(NO NUMBER, NAME VARCHAR2(20));  //컬럼의 이름(NO) 타입(NUMBER)
    
테이블이 생성되었습니다.
    
-- 테이블 행추가
    
SQL> INSERT INTO TEST VALUES(1,’박상은’);
    
1 개의 행이 만들어졌습니다.

SQL> INSERT INTO TEST(NO, NAME) VALUES(2,’PSE’);
   
1 개의 행이 만들어졌습니다.
 
-- 조회

SQL> SELECT * FROM TEST;

        NO NAME
---------- ----------------------------------------
         1 박상은
         2 PSE

SQL> SELECT * FROM TEST WHERE NO=1;

            NO NAME
---------- ----------------------------------------
         1 박상은

-- 삭제
    
SQL> DROP TABLE TEST;

테이블이 삭제되었습니다.

-- 종료
    
SQL> EXIT;
